#==========================================================
# Tree coefficient
#
# r_{2k+1}(t)
#==========================================================

def r_coeff(k,t):

    factorial = math.factorial(2*k+1)

    def integrand(u):

        return (
            poch(1-u,2*k+1)
            *
            zeta(2*k+2-u)
        )

    value,error = quad(
        integrand,
        0.0,
        t,
        epsabs=PARAM.epsabs,
        epsrel=PARAM.epsrel,
        limit=PARAM.quad_limit
    )

    return -2.0*value/factorial


#==========================================================
# derivative
#
# dr/dt
#==========================================================

def dr_coeff(k, t):

    factorial = math.factorial(2*k+1)

    a = 1.0 - t

    #
    # Gamma pole
    #
    if abs(a-round(a)) < 1e-12 and a <= 0:

        return 0.0

    value = (
        -2.0
        * poch(a, 2*k+1)
        * zeta(2*k+2-t)
        / factorial
    )

    if not np.isfinite(value):

        return 0.0

    return value


#==========================================================
# coefficient cache

_TREE_CACHE={}


#==========================================================
# Build coefficients
#==========================================================

def build_tree_coefficients(t,
                            order=None):

    if order is None:

        order=PARAM.TREE_ORDER

    key=(float(t),order)

    if key in _TREE_CACHE:

        return _TREE_CACHE[key]

    coeff=np.zeros(order)

    for k in range(order):

        coeff[k]=r_coeff(k,t)

    _TREE_CACHE[key]=coeff

    return coeff

EPS = 1.0e-12
EPS_SERIES = 1.0e-8


#==========================================================
# Stable phi(x)
#
#      x e^x
# φ = --------
#      e^x-1
#
#==========================================================

def phi_exprel(x):

    x = np.complex128(x)

    if abs(x) < EPS_SERIES:

        x2 = x*x
        x4 = x2*x2

        return (
            1.0
            + x/2.0
            + x2/12.0
            - x4/720.0
        )

    return x*np.exp(x)/np.expm1(x)


#==========================================================
# Principal Loop Function
#
#          (-z)^t-1
# P(t,z)=------------
#         z log(-z)
#
#==========================================================

def P(t, z):

    z = np.complex128(z)

    L = clog(-z)

    if abs(z) < EPS:
        raise ValueError("P(t,z): z too close to 0.")

    x = t*L

    numerator = np.expm1(x)
    denominator = z*L

    if abs(denominator) < EPS:
        raise ValueError(
            "P(t,z): evaluation too close to branch point."
        )

    return numerator/denominator


#==========================================================
# Tree Series
#
# R(t,z)
#
#==========================================================

def R(t,
      z,
      order=None):

    if order is None:
        order = PARAM.TREE_ORDER

    coeff = build_tree_coefficients(
        t,
        order
    )

    value = 0.0 + 0.0j
    power = z

    for c in coeff:

        value += c*power
        power *= z*z

    return value


#==========================================================
# Completed Function
#
# F=P exp(R)
#
#==========================================================

def Completed(
        t,
        z,
        order=None):

    return (
        P(t, z)
        *
        np.exp(
            R(
                t,
                z,
                order
            )
        )
    )


#==========================================================
# d(log P)/dz
#
#==========================================================

def dlogP_dz(
        t,
        z):

    z = np.complex128(z)

    if abs(z) < EPS:
        raise ValueError(
            "dlogP_dz: z too close to 0."
        )

    L = clog(-z)

    if abs(L) < EPS:
        raise ValueError(
            "dlogP_dz: too close to branch point z=-1."
        )

    x = t*L

    phi = phi_exprel(x)

    return (

        phi/(z*L)

        -

        1.0/z

        -

        1.0/(z*L)

    )


#==========================================================
# d(log P)/dt
#
#==========================================================

def dlogP_dt(
        t,
        z):

    z = np.complex128(z)

    L = clog(-z)

    if abs(t) < EPS:
        return np.inf

    x = t*L

    phi = phi_exprel(x)

    return phi/t


#==========================================================
# dR/dz
#
#==========================================================

def dR_dz(
        t,
        z,
        order=None):

    if order is None:
        order = PARAM.TREE_ORDER

    coeff = build_tree_coefficients(
        t,
        order
    )

    value = 0.0 + 0.0j
    power = 1.0 + 0.0j

    for k, c in enumerate(coeff):

        value += (2*k+1)*c*power
        power *= z*z

    return value


#==========================================================
# dR/dt
#
#==========================================================

def dR_dt(
        t,
        z,
        order=None):

    if order is None:
        order = PARAM.TREE_ORDER

    coeff = build_dr_coefficients(
        t,
        order
    )

    value = 0.0 + 0.0j
    power = z

    for c in coeff:

        value += c*power
        power *= z*z

    return value


#==========================================================
# Completed logarithmic derivatives
#
#==========================================================

def dlogF_dz(
        t,
        z):

    return (

        dlogP_dz(t, z)

        +

        dR_dz(t, z)

    )


def dlogF_dt(
        t,
        z):

    return (

        dlogP_dt(t, z)

        +

        dR_dt(t, z)

    )

def omega_z(t,z):

# NOTE:
# This computes numerical consistency of
# mixed partial derivatives only.
# It is not a nontrivial curvature test.

#==========================================================
# Numerical derivatives
#==========================================================

def partial_z(func,
              t,
              z,
              h=1e-6):

    return (

        func(t,z+h)

        -

        func(t,z-h)

    )/(2*h)


def partial_t(func,
              t,
              z,
              h=1e-6):

    return (

        func(t+h,z)

        -

        func(t-h,z)

    )/(2*h)


#==========================================================
# Curvature
#
# Ω=dω
#
#==========================================================

def curvature(t,
              z):

    return (

        partial_z(
            omega_t,
            t,
            z
        )

        -

        partial_t(
            omega_z,
            t,
            z
        )

    )


#==========================================================
# Curvature norm
#==========================================================

def curvature_norm(t,z):

    return abs(
        curvature(
            t,
            z
        )
    )

#==========================================================
# Circular loop
#==========================================================

def circle_loop(center=-1.0,
                radius=1.0e-4,
                points=4000):

    theta=np.linspace(
        0.0,
        2.0*np.pi,
        points,
        endpoint=True
    )

    z=center+radius*np.exp(1j*theta)

    return theta,z


#==========================================================
# Line integral
#==========================================================

def line_integral(path,
                  t):

    integral=0.0+0.0j

    for i in range(len(path)-1):

        dz=path[i+1]-path[i]

        integral+=omega_z(
            t,
            path[i]
        )*dz

    return integral


#==========================================================
# Loop integral
#==========================================================

def loop_integral(
        t,
        center=-1.0,
        radius=1e-4,
        points=4000):

    _,path=circle_loop(
        center=center,
        radius=radius,
        points=points
    )

    return line_integral(
        path,
        t
    )

#==========================================================
# Canonical skew generator
#==========================================================

def build_loop_generator(
        order=None,
        beta=None):

    if order is None:

        order=PARAM.TREE_ORDER

    if beta is None:

        beta=PARAM.beta

    G=np.zeros(
        (order,order)
    )

    #
    # nearest neighbour
    #

    for k in range(order-1):

        c=beta*np.sqrt(k+1)

        G[k,k+1]= c

        G[k+1,k]=-c

    return G


#==========================================================
# Loop operator
#
# M = H exp(G)
#==========================================================

def build_loop_operator(
        t,
        order=None):

    if order is None:

        order=PARAM.TREE_ORDER

    H=holonomy(t)

    G=build_loop_generator(
        order
    )

    M=H*expm(G)

    return M,G


#==========================================================
# Unitary error
#==========================================================

def unitary_error(M):

    I=identity(
        M.shape[0]
    )

    return np.linalg.norm(

        M.conjugate().T

        @

        M

        -

        I

    )

#==========================================================
# Tree operator
#==========================================================

def build_tree_operator(
        t,
        order=None,
        alpha=None):

    if order is None:
        order = PARAM.TREE_ORDER

    if alpha is None:
        alpha = PARAM.alpha

    d = build_dr_coefficients(
        t,
        order
    )

    L = np.zeros(
        (order, order),
        dtype=float
    )

    #
    # diagonal
    #

    for k in range(order):
        L[k, k] = -d[k]

    #
    # nearest neighbour coupling
    #

    for k in range(order - 1):

        c = alpha*np.sqrt(k + 1)

        L[k, k + 1] = c
        L[k + 1, k] = c

    return L


#==========================================================
# Stopping operator
#
# shift=None
#     -> no artificial kernel
#
# shift=value
#     -> L - shift I
#
#==========================================================

def build_stopping_operator(
        t,
        order=None,
        shift=None):

    L = build_tree_operator(
        t,
        order
    )

    if shift is None:
        shift = 0.0

    Lstop = (
        L
        -
        shift*np.identity(
            L.shape[0]
        )
    )

    return Lstop


#==========================================================
# Optional:
# produce a kernel intentionally
#==========================================================

def shift_to_ground_state(
        L):

    vals = np.linalg.eigvalsh(L)

    lam = np.min(vals)

    Lstop = (
        L
        -
        lam*np.identity(
            L.shape[0]
        )
    )

    return Lstop, lam


#==========================================================
# Kernel dimension
#==========================================================

def kernel_dimension(
        L):

    vals = np.linalg.eigvalsh(L)

    return int(
        np.sum(
            np.abs(vals)
            <
            PARAM.tol
        )
    )


#==========================================================
# Spectral gap
#==========================================================

def spectral_gap(
        L):

    vals = np.linalg.eigvalsh(L)

    vals = np.sort(vals)

    nz = vals[
        np.abs(vals)
        >
        PARAM.tol
    ]

    if len(nz) == 0:
        return 0.0

    return np.min(
        np.abs(nz)
    )


#==========================================================
# Kernel projector
#==========================================================

def kernel_projector(
        L):

    vals, vecs = np.linalg.eigh(L)

    P = np.zeros_like(
        L,
        dtype=float
    )

    for i, v in enumerate(vals):

        if abs(v) < PARAM.tol:

            u = vecs[:, i]

            P += np.outer(
                u,
                u
            )

    return P


#==========================================================
# Moore-Penrose inverse
#==========================================================

def stopping_pinv(
        L):

    vals, vecs = np.linalg.eigh(L)

    Pinv = np.zeros_like(
        L,
        dtype=float
    )

    for i, v in enumerate(vals):

        if abs(v) > PARAM.tol:

            u = vecs[:, i]

            Pinv += (
                1.0/v
            ) * np.outer(
                u,
                u
            )

    return Pinv

#==========================================================
# Commutator
#==========================================================

def commutator(A, B):

    return A @ B - B @ A


#==========================================================
# Conjugation
#==========================================================

def conjugate(M, A):

    return M.conjugate().T @ A @ M


#==========================================================
# Kernel preservation
#==========================================================

def kernel_preservation(L, M):

    P = kernel_projector(L)

    P2 = conjugate(M, P)

    return np.linalg.norm(P - P2)


#==========================================================
# Operator preservation
#==========================================================

def operator_preservation(L, M):

    L2 = conjugate(M, L)

    return np.linalg.norm(L - L2)

def bianchi_defect(
        L,
        G):

    C1 = commutator(
        L,

    )

    return np.linalg.norm(
        C1 + C2 + C3

#==========================================================
# Renormalized d(log F)/dt

def dlogF_dt_ren(
        t,
        z):

    wt = dlogF_dt(
        t,
        z

    if abs(t) < PARAM.tol:

        # finite part

        L = clog(-z)

        return 0.5*L

    return wt - 1.0/t

#==========================================================
# Gauge connection
#
# A = Az dz + At dt
#
# Az = wz I + alpha Gamma
# At = wt I + beta L
#==========================================================

def build_gauge_connection(
        t,
        z,
        alpha=1.0,
        beta=1.0):

    wz = dlogF_dz(
        t,
        z
    )

    wt = dlogF_dt_ren(
        t,
        z
    )

    Gamma = build_connection_matrix(
        t
    )

    L = build_stopping_operator(
        t
    )

    n = L.shape[0]

    I = np.identity(
        n,
        dtype=complex
    )

    Az = (
        wz*I
        +
        alpha*Gamma
    )

    At = (
        wt*I
        +
        beta*L
    )

    return Az, At, L, Gamma

#==========================================================
# Stopping kernel vector
#==========================================================

def stopping_kernel_vector(
        L):

    vals, vecs = np.linalg.eigh(
        L
    )

    i0 = np.argmin(
        np.abs(vals)
    )

    lam0 = vals[i0]

    u0 = vecs[:, i0]

    return lam0, u0


#==========================================================
# Curvature action
#==========================================================

def curvature_action(
        L,
        F):

    lam0, u0 = (
        stopping_kernel_vector(
            L
        )
    )

    Fu = F @ u0

    return (
        lam0,
        u0,
        Fu
    )


#==========================================================
# Coupling measures
#==========================================================

def curvature_coupling(
        L,
        F):

    lam0, u0, Fu = (
        curvature_action(
            L,
            F
        )
    )

    #
    # norm
    #

    nrm = np.linalg.norm(
        Fu
    )

    #
    # expectation
    #

    amp = np.vdot(
        u0,
        Fu
    )

    #
    # angle
    #

    if nrm < PARAM.tol:

        angle = 0.0

    else:

        c = (

            np.abs(amp)

            /

            nrm

        )

        c = np.clip(
            c,
            0.0,
            1.0
        )

        angle = np.arccos(
            c
        )

    return (
        nrm,
        amp,
        angle
    )

#==========================================================
# Stopping projector

def stopping_projector(
        u0):

    u0 = np.asarray(
        u0,
        dtype=complex

    nrm = np.linalg.norm(
        u0

            "kernel vector is zero"
        )

    u0 = u0 / nrm

    return np.outer(
        u0,
        u0.conjugate()
    )


#==========================================================
# Leakage by Gamma
#==========================================================

def gamma_leakage(
        Gamma,
        u0):

    n = len(u0)

    P0 = stopping_projector(
        u0
    )

    Q0 = (
        np.identity(
            n
        )
        -
        P0
    )

    Gu = Gamma @ u0

    leak = np.linalg.norm(
        Q0 @ Gu
    )

    parallel = np.linalg.norm(
        P0 @ Gu
    )

    return (
        leak,
        parallel,
        Gu
    )

#==========================================================
# Numerical derivative of L

def dL_dt(
        t,
        h=1.0e-5):

    Lp = build_stopping_operator(
        t + h
    )

    Lm = build_stopping_operator(
        t - h
    )

    return (
        Lp - Lm
    )/(2.0*h)


#==========================================================
# Kernel vector
#==========================================================

def stopping_kernel_vector(
        L):

    vals, vecs = np.linalg.eigh(
        L
    )

    i0 = np.argmin(
        np.abs(vals)
    )

    u0 = vecs[:, i0]

    u0 = (
        u0
        /
        np.linalg.norm(u0)
    )

    return vals[i0], u0


#==========================================================
# Spectral gap
#==========================================================

def stopping_gap(
        L):

    vals = np.sort(
        np.linalg.eigvalsh(L)
    )

    nz = vals[
        np.abs(vals)
        > PARAM.tol
    ]

    if len(nz) == 0:
        return 0.0

    return np.min(
        np.abs(nz)
    )

    return (

        L,
        dL

#==========================================================
# Kernel projector
#
# P = uu*
#==========================================================

def kernel_projector_bundle(
        u):

    u = np.asarray(
        u,
        dtype=complex
    ).reshape(-1)

    nrm = np.linalg.norm(
        u
    )

    if nrm < PARAM.tol:

        raise ValueError(
            "kernel vector is zero"
        )

    u = u / nrm

    return np.outer(
        u,
        u.conjugate()
    )


#==========================================================
# Projector derivative
#==========================================================

def projector_derivative(
        t,
        h=1.0e-5):

    Lp = build_stopping_operator(t + h)
    _, up = stopping_kernel_vector(Lp)

    Lm = build_stopping_operator(t - h)
    _, um = stopping_kernel_vector(Lm)

    #
    # phase fixing
    #
    if np.vdot(up, um).real < 0:
        up = -up

    Pp = kernel_projector_bundle(up)
    Pm = kernel_projector_bundle(um)

    return (Pp - Pm)/(2*h)


#==========================================================
# Bundle curvature
#
# K=[Pdot,P]
#==========================================================

def stopping_bundle_curvature(
        t):

    L = build_stopping_operator(
        t
    )

    _, u = (
        stopping_kernel_vector(
            L
        )
    )

    P = kernel_projector_bundle(
        u
    )

    Pdot = projector_derivative(
        t
    )

    K = commutator(
        Pdot,
        P
    )

    return K, P, Pdot

#==========================================================
# Ground stopping vector
#==========================================================

def stopping_ground_vector(t):

    L = build_stopping_operator(t)

    vals, vecs = np.linalg.eigh(L)

    i0 = np.argmin(vals)

    lam0 = vals[i0]

    u0 = vecs[:, i0]

    return lam0, u0


#==========================================================
# Kernel projector
#==========================================================

def kernel_bundle_projector(t):

    _, u = stopping_ground_vector(t)

    return np.outer(
        u,
        u.conjugate()
    )


#==========================================================
# Projector derivative
#==========================================================

def projector_derivative(
        t,
        h=DIFF_STEP):

    Pp = kernel_bundle_projector(
        t + h
    )

    Pm = kernel_bundle_projector(
        t - h
    )

    return (
        Pp - Pm
    )/(2.0*h)


#==========================================================
# Bundle curvature
#
# K=[P',P]
#==========================================================

def bundle_curvature(
        t,
        h=DIFF_STEP):

    P = kernel_bundle_projector(t)

    dP = projector_derivative(
        t,
        h
    )

    return commutator(
        dP,
        P
    )


#==========================================================
# Curvature norm
#==========================================================

def bundle_curvature_norm(
        t):

    K = bundle_curvature(t)

    return np.linalg.norm(K)

# Curvature flux
#
# Φ(T)=∫ ||K|| dt

def curvature_flux(

    vals = np.array(

    return np.trapz(
        vals,
        ts
    )


#==========================================================
# Holonomy overlap
#==========================================================

def kernel_overlap(
        t0,
        t1):

    _, u0 = stopping_ground_vector(t0)
    _, u1 = stopping_ground_vector(t1)

    amp = np.vdot(
        u0,
        u1
    )

    return amp


#==========================================================
# Holonomy displacement
#==========================================================

def kernel_displacement(
        t0,
        t1):

    amp = kernel_overlap(
        t0,
        t1
    )

    c = np.abs(amp)

    c = np.clip(
        c,
        0.0,
        1.0
    )

    return np.sqrt(
        1.0 - c*c
    )

# Berry phase
#==========================================================

def berry_phase(
        t0,
        t1):

    amp = kernel_overlap(
        t0,
        t1
    )

    return np.angle(amp)

#==========================================================

def holonomy_prediction(

    Phi = curvature_flux(

#==========================================================
# Gauge-fixed ground vector
#==========================================================

def ground_vector(t):

    _, u = stopping_ground_vector(t)

    return u


#==========================================================
# Parallel transport gauge
#==========================================================

def gauge_fixed_vector(
        tref,
        t):

    u0 = ground_vector(tref)
    u = ground_vector(t)

    amp = np.vdot(
        u0,
        u
    )

    if np.abs(amp) > PARAM.tol:

        u = (
            amp.conjugate()
            /
            np.abs(amp)
        ) * u

    return u


#==========================================================
# Numerical derivative
#==========================================================

def ground_derivative(
        tref,
        t,
        h=BERRY_STEP):

    up = gauge_fixed_vector(
        tref,
        t + h
    )

    um = gauge_fixed_vector(
        tref,
        t - h
    )

    return (
        up - um
    )/(2.0*h)


#==========================================================
# Berry connection
#
# A=i<u,u'>
#==========================================================

def berry_connection(
        tref,
        t):

    u = gauge_fixed_vector(
        tref,
        t
    )

    du = ground_derivative(
        tref,
        t
    )

    return np.real_if_close(
        1j*np.vdot(
            u,
            du
        )
    )


#==========================================================
# Berry curvature
#
# Ω=dA/dt
#==========================================================

def berry_curvature(
        tref,
        t,
        h=BERRY_STEP):

    Ap = berry_connection(
        tref,
        t + h
    )

    Am = berry_connection(
        tref,
        t - h
    )

    return (
        Ap - Am
    )/(2.0*h)

# Berry phase
#==========================================================

def berry_phase_integral(
        t0,
        t1,
        samples=400):

    ts = np.linspace(

    vals = np.array(
        [
            berry_connection(
                t0,
                tt
            )
            for tt in ts
        ]
    )

    return np.trapezoid(
        vals,
        ts
    )

def topological_charge(

def spectral_gap(t):

#==========================================================
# Image projector
#
# P_im = I - Pker
#==========================================================

def image_projector(
        L):

    n = L.shape[0]

    Pker = kernel_projector(
        L
    )

    return (
        np.identity(
            n
        )
        -
        Pker
    )


#==========================================================
# Hodge decomposition
#
# x = Pker x + Pim x
#==========================================================

def hodge_decomposition(
        L,
        x):

    Pker = kernel_projector(
        L
    )

    Pim = image_projector(
        L
    )

    xker = Pker @ x

    xim = Pim @ x

    return (
        xker,
        xim,
        Pker,
        Pim
    )


#==========================================================
# Orthogonality
#==========================================================

def orthogonality_error(
        Pker,
        Pim):

    return np.linalg.norm(
        Pker @ Pim
    )


#==========================================================
# Completeness
#
# Pker+Pim = I
#==========================================================

def completeness_error(
        Pker,
        Pim):

    I = np.identity(
        Pker.shape[0]
    )

    return np.linalg.norm(

        Pker

        +

        Pim

        -

        I

    )


#==========================================================
# Reconstruction
#
# x = xker + xim
#==========================================================

def reconstruction_error(
        x,
        xker,
        xim):

    return np.linalg.norm(

        x

        -

        xker

        -

        xim

    )


#==========================================================
# Hodge energy
#==========================================================

def hodge_energy(
        xker,
        xim):

    return (

        np.linalg.norm(xker)**2,

        np.linalg.norm(xim)**2

    )

#==========================================================
# Rough Laplacian
#
# Δ0 = Γ*Γ
#==========================================================

def rough_laplacian(
        Gamma):

    return (
        Gamma.conjugate().T
        @
        Gamma
    )


#==========================================================
# Curvature term
#
# R = [Γ,L]
#==========================================================

def curvature_term(
        t):

    Gamma = build_connection_matrix(
        t
    )

    L = build_stopping_operator(
        t
    )

    return commutator(
        Gamma,
        L
    )


#==========================================================
# Weitzenböck operator
#
# ΔW = Γ*Γ + R
#==========================================================

def weitzenbock_operator(
        t):

    Gamma = build_connection_matrix(
        t
    )

    Delta0 = rough_laplacian(
        Gamma
    )

    R = curvature_term(
        t
    )

    DeltaW = (
        Delta0
        +
        R
    )

    return (
        DeltaW,
        Delta0,
        R,
        Gamma
    )


#==========================================================
# Numerical Laplacian
#

def canonical_rough_laplacian(
        At):

    return (
        At.conjugate().T
        @

    )

# Curvature

# [At,L]

def canonical_curvature(
        At,
        L):

    return (
        At @ L
        -
        L @ At
    )

def canonical_weitzenbock(

    )

    DeltaA = (
        Delta0
        +
        R
    )

    return (
        DeltaA,
        Delta0,
        R,
        At,
        L
    )

def continuous_sign_phase(
        t0,
        t1,
        steps=2000):

    ts = np.linspace(
        t0,
        t1,
        steps
    )

    L0 = build_stopping_operator(
        ts[0]
    )

    _, u_prev = stopping_kernel_vector(
        L0
    )

    u_init = u_prev.copy()

    flips = 0

    for tt in ts[1:]:

        L = build_stopping_operator(
            tt
        )

        _, u = stopping_kernel_vector(
            L
        )

        ov = np.vdot(
            u_prev,
            u
        )

        if ov.real < 0:

            u = -u
            flips += 1

        u_prev = u

    phase = np.angle(
        np.vdot(
            u_init,
            u_prev
        )
    )

    print()
    print("Phase")
    print(phase)

    print()
    print("Sign flips")
    print(flips)

    return phase, flips

def stopping_criticality_scan(
        tmin=-2.0,
        tmax=2.0,
        samples=400,
        verbose=False):

    ts = np.linspace(
        tmin,
        tmax,
        samples
    )

    gaps = []
    at_norms = []
    delta_norms = []
    curvatures = []

    for t in ts:

        gap = np.nan
        at_norm = np.nan
        delta_norm = np.nan
        curvature_norm = np.nan

        try:

            #--------------------------------------
            # stopping operator
            #--------------------------------------
            L = build_stopping_operator(t)

            vals, vecs = np.linalg.eigh(L)

            idx = np.argsort(vals)
            vals = vals[idx]
            vecs = vecs[:, idx]

            u0 = vecs[:, 0]

            #--------------------------------------
            # spectral gap
            #--------------------------------------
            if len(vals) >= 2:
                gap = vals[1] - vals[0]

            #--------------------------------------
            # canonical connection
            #--------------------------------------
            At, Ltmp, dL = stopping_connection(t)

            at_norm = np.linalg.norm(At)

            #--------------------------------------
            # curvature
            #--------------------------------------
            Curv = canonical_curvature(
                At,
                Ltmp
            )

            curvature_norm = np.linalg.norm(
                Curv
            )

            #--------------------------------------
            # canonical Weitzenbock operator
            #--------------------------------------
            Delta0 = canonical_rough_laplacian(
                At
            )

            DeltaA = (
                Delta0
                +
                Curv
            )

            delta_norm = np.linalg.norm(
                DeltaA @ u0
            )

        except Exception as e:

            if verbose:
                print()
                print("Exception at t =", t)
                print(type(e))
                print(e)

        #--------------------------------------
        # append exactly once
        #--------------------------------------
        gaps.append(gap)
        at_norms.append(at_norm)
        delta_norms.append(delta_norm)
        curvatures.append(curvature_norm)

    gaps = np.array(gaps)
    at_norms = np.array(at_norms)
    delta_norms = np.array(delta_norms)
    curvatures = np.array(curvatures)

    #--------------------------------------
    # sanity check
    #--------------------------------------
    print()
    print("Lengths")
    print(
        len(ts),
        len(gaps),
        len(at_norms),
        len(delta_norms),
        len(curvatures)
    )

    #--------------------------------------
    # Report
    #--------------------------------------
    print()
    print("Minimum gap")
    print(np.nanmin(gaps))

    print()
    print("Maximum gap")
    print(np.nanmax(gaps))

    print()
    print("Maximum ||At||")
    print(np.nanmax(at_norms))

    print()
    print("Maximum ||ΔAu0||")
    print(np.nanmax(delta_norms))

    print()
    print("Maximum curvature")
    print(np.nanmax(curvatures))

    #--------------------------------------
    # Plot
    #--------------------------------------
    fig, ax = plt.subplots(
        figsize=(10, 6)
    )

    ax.plot(
        ts,
        gaps,
        label="Gap"
    )

    ax.plot(
        ts,
        at_norms,
        label="||At||"
    )

    ax.plot(
        ts,
        delta_norms,
        label="||ΔAu0||"
    )

    ax.plot(
        ts,
        curvatures,
        label="Curvature"
    )

    ax.set_xlabel("t")
    ax.set_ylabel("Magnitude")
    ax.set_title(
        "Stopping Criticality"
    )

    ax.grid(True)
    ax.legend()

    plt.show()

    return (
        ts,
        gaps,
        at_norms,
        delta_norms,
        curvatures
    )

def fit_gap_scaling(
        gaps,
        values,
        vmax=np.inf):

    mask = (
        np.isfinite(gaps)
        &
        np.isfinite(values)
        &
        (gaps > 1e-12)
        &
        (values > 1e-12)
        &
        (values < vmax)
    )

    if np.sum(mask) < 5:
        print()
        print("Not enough data.")
        return np.nan

    x = np.log(gaps[mask])
    y = np.log(values[mask])

    a, b = np.polyfit(
        x,
        y,
        1
    )

    alpha = -a

    print()
    print("Scaling exponent α")
    print(alpha)

    plt.figure(
        figsize=(6, 5)
    )

    plt.scatter(
        x,
        y,
        s=12
    )

    xx = np.linspace(
        x.min(),
        x.max(),
        200
    )

    plt.plot(
        xx,
        a * xx + b
    )

    plt.xlabel("log gap")
    plt.ylabel("log value")
    plt.grid(True)
    plt.show()

    return alpha
