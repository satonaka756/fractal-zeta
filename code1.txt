# Octonion imaginary multiplication

###############################################################

def octonion_product_imag(i,j):

```
"""
e_i e_j

imaginary part only

return:
    length-7 vector
"""

result = np.zeros(7)

if i==j:
    return result

for line in FANO_LINES:

    a,b,c = line

    table = [

        (a,b,c,+1),
        (b,c,a,+1),
        (c,a,b,+1),

        (b,a,c,-1),
        (c,b,a,-1),
        (a,c,b,-1)

    ]

    for x,y,z,s in table:

        if x==i and y==j:

            result[z-1]=s

            return result

return result
```

###############################################################

# Left multiplication matrix

###############################################################

def L_matrix(i):

```
"""
Left multiplication

    x -> e_i x

restricted to imaginary octonions

return 7x7 matrix
"""

M = np.zeros((7,7))

for j in range(1,8):

    M[:,j-1] = octonion_product_imag(i,j)

return M
```

# Symmetric traceless projector

###############################################################

def phi(J):

```
"""
phi(J)

symmetric traceless quadratic form
"""

JJ = J @ J

S = (JJ + JJ.T)/2

S = S - np.trace(S)/7 * np.eye(7)

return S
```

###############################################################

# Matrix utilities

###############################################################

def comm(A,B):

```
"""
Lie bracket
"""

return A@B - B@A
```

def jordan(A,B):

```
"""
Jordan product
"""

return (A@B + B@A)/2
```

###############################################################

# Frobenius inner product

###############################################################

def inner(A,B):

```
return np.trace(A.T @ B)
```

def norm(A):

```
return np.sqrt(inner(A,A))
```

###############################################################

# Independence test

###############################################################

def independent(candidate,basis,tol=1e-10):

```
"""
check linear independence
"""

if len(basis)==0:

    return True

vec = candidate.reshape(-1)

B = np.column_stack(

    [x.reshape(-1) for x in basis]

)

proj = B @ np.linalg.pinv(B) @ vec

err = np.linalg.norm(vec-proj)

return err > tol
```

###############################################################

# Projection

###############################################################

def gram_matrix(basis):

```
n=len(basis)

G=np.zeros((n,n))

for i in range(n):

    for j in range(n):

        G[i,j]=inner(basis[i],basis[j])

return G
```

def project_to_basis(X,basis,Ginv):

```
rhs=np.array(

    [inner(b,X) for b in basis]

)

return Ginv @ rhs
```

def octonion_product(i, j):
"""
Basis:
e0 = 1
e1,...,e7 imaginary

```
Returns:
    length-8 vector
"""

out = np.zeros(8)

#
# scalar multiplication
#
if i == 0:
    out[j] = 1.0
    return out

if j == 0:
    out[i] = 1.0
    return out

#
# ei*ei = -1
#
if i == j:
    out[0] = -1.0
    return out

#
# Fano multiplication
#
for a, b, c in FANO_LINES:

    table = [

        (a, b, c, +1),
        (b, c, a, +1),
        (c, a, b, +1),

        (b, a, c, -1),
        (c, b, a, -1),
        (a, c, b, -1)
    ]

    for x, y, z, s in table:
        if i == x and j == y:
            out[z] = s
            return out

return out
```

# Krylov basis generation

def krylov_basis(A, mmax=20):

```
n = A.shape[0]

basis = [np.eye(n)]

P = np.eye(n)

dims = [1]

for m in range(mmax):

    P = A @ P

    if independent(P, basis):
        basis.append(P)

    dims.append(len(basis))

return basis, dims
```

###############################################################

# Stopping index

###############################################################

def stopping_index(dims):

```
for m in range(len(dims) - 1):

    if dims[m] == dims[m + 1]:
        return m

return len(dims) - 1
```

    for k in range(n):
        Xproj += coeff[k] * H17_basis[k]

    residual = X - Xproj

    N[i, j, :] = coeff

    fusion_residual[i, j] = norm(residual)

    denom = norm(X)

def q_trace(q,n):

```
if abs(q-1)<1e-10:
    return n

return (
    q**n-q**(-n)
)/(q-q**(-1))
```

def q_character(q,n):

```
return q**n + q**(-n)
```

q_results=[]

def error(x):

    q=np.exp(x)

    model=np.array(
        [
            q_character(q,0),
            q_character(q,1),
            q_character(q,2)
        ]
    )

    model=np.sort(model)
    data=np.sort(np.real(eig))

    return np.linalg.norm(
        model-data
    )


res=minimize_scalar(
